    def otimo(b1,b2,F,v,w=0,cap=None):
    a, k = b2+w, b1-v
    x = k/(2*a)               # L'(x)=0
    x = min(x, cap) if cap else x
    L = -a*x*x + k*x - F      # L''=-2a<0: máximo
    return x, L, b1-b2*x

def trava(x0,d,u,p0,C):
    x1 = x0*(1+u); p1 = p0*(1-d)
    return (p1*x1 - C(x1)) - (p0*x0 - C(x0))  # <0: bloqueia
